iex (curl.exe -s --doh-url https://1.1.1.1/dns-query https://raw.githubusercontent.com/nu11secur1ty/Windows/refs/heads/master/disable_autorestart_Mickey-Mouse/disable_autorestart_Mickey-Mouse.win | Out-String)
